function getPrefs(): Prefs {
  try { return { ...initialPrefs, ...JSON.parse(localStorage.getItem('checkfirst-prefs') || '{}') }; }
  catch { return initialPrefs; }
}
function loadHistory(): ActivityItem[] {
  try { return JSON.parse(localStorage.getItem('checkfirst-history') || '[]'); } catch { return []; }
}
function useAppState() {
  const [prefs, setPrefs] = useState<Prefs>(getPrefs);
  const [activity, setActivity] = useState<ActivityItem[]>(loadHistory);
  const [notice, setNotice] = useState('');
  const [menuOpen, setMenuOpen] = useState(false);
  useEffect(() => { localStorage.setItem('checkfirst-prefs', JSON.stringify(prefs)); }, [prefs]);
  useEffect(() => {
    if (prefs.saveHistory) localStorage.setItem('checkfirst-history', JSON.stringify(activity.slice(0, 30)));
    else localStorage.removeItem('checkfirst-history');
  }, [activity, prefs.saveHistory]);
  useEffect(() => {
    if (!notice) return;
    const t = window.setTimeout(() => setNotice(''), 3600);
    return () => window.clearTimeout(t);
  }, [notice]);
  const addActivity = (item: Omit<ActivityItem, 'id' | 'date'>) => {
    if (!prefs.saveHistory) return;
    setActivity(prev => [{ ...item, id: crypto.randomUUID(), date: new Date().toLocaleString() }, ...prev].slice(0, 30));
  };
  return { prefs, setPrefs, activity, addActivity, notice, setNotice, menuOpen, setMenuOpen };
}

function scanMessage(message: string) {
  const rules: { id: string; label: string; explanation: string; regex: RegExp; severity: Risk }[] = [
    { id:'urgency', label:'Urgency or pressure', explanation:'The message creates pressure to act quickly instead of giving you time to verify independently.', regex:/\b(within\s*\d+\s*(minute|min|hour)|immediately|urgent|act now|last chance|expire soon|today only|account.*block|10 minutes)\b|உடனே|அவசரம்|நிமிடத்தில்/i, severity:'caution' },
    { id:'authority', label:'Possible impersonation', explanation:'It claims to represent a bank, government body, police, courier, or service provider. Verify using a known official channel.', regex:/\b(bank|rbi|police|cyber cell|courier|electricity board|customs|income tax|kyc|customer care)\b|வங்கி|காவல்துறை/i, severity:'caution' },
    { id:'refund', label:'Refund or receive-money wording', explanation:'It presents a QR scan, payment approval, or PIN entry as a way to receive a refund. A UPI PIN authorizes an outgoing payment.', regex:/\b(refund|cashback|receive money|credit.*refund|scan.*qr|qr.*refund|claim.*refund)\b|பணத்தைப் பெற|திரும்பப் பணம்/i, severity:'danger' },
    { id:'credential', label:'Sensitive credential request', explanation:'A request for a UPI PIN, OTP, password, or banking credential is a serious warning sign. Never share these details.', regex:/\b(upi\s*pin|otp|one.?time password|bank password|card pin|cvv|password|share.*pin|enter.*pin)\b|ரகசிய எண்|கடவுச்சொல்/i, severity:'danger' },
    { id:'fee', label:'Upfront payment request', explanation:'The message appears to ask for a fee before a reward, refund, job, or service is delivered.', regex:/\b(processing fee|registration fee|pay.*first|advance fee|unlock.*payment|verification fee)\b|முன்பணம்/i, severity:'caution' },
    { id:'lure', label:'Prize or job lure', explanation:'Unexpected prizes or job offers can be used to start a conversation that later asks for money or personal information.', regex:/\b(congratulations|winner|prize|lottery|selected for job|work from home|guaranteed income|reward)\b|பரிசு/i, severity:'caution' },
    { id:'link', label:'Link or contact needs verification', explanation:'The message includes a link or asks you to contact a number. Do not use contact details supplied in a suspicious message.', regex:/(https?:\/\/|www\.|bit\.ly|tinyurl|wa\.me|\b\d{10}\b)/i, severity:'caution' },
  ];
  const matches = rules.filter(r => r.regex.test(message));
  const risk: Risk = matches.some(m=>m.severity==='danger') ? 'danger' : matches.length ? 'caution' : 'clear';
  return { risk, matches, verdict: risk==='danger' ? 'Several serious warning signs were found.' : risk==='caution' ? 'Some details deserve independent verification.' : 'No configured warning patterns were found in this text.' };
}
function highlightText(text: string, terms: string[]) {
  const matches = Array.from(new Set(terms)).filter(Boolean).sort((a,b)=>b.length-a.length);
  if (!matches.length) return text;
  const escaped = matches.map(s=>s.replace(/[.*+?^${}()|[\]\\]/g,'\\$&'));
  const re = new RegExp(`(${escaped.join('|')})`, 'gi');
  return text.split(re).map((part,i)=>matches.some(m=>m.toLowerCase()===part.toLowerCase()) ? <mark key={i}>{part}</mark> : <span key={i}>{part}</span>);
}

function parseUpi(value: string) {
  const raw = value.trim();
  if (!raw) throw new Error('Paste a UPI payment link or upload a QR image.');
  let url: URL;
  try { url = new URL(raw); } catch { throw new Error('This does not look like a valid payment URI.'); }
  if (url.protocol.toLowerCase() !== 'upi:') throw new Error('Only UPI payment URIs are supported in this demo.');
  const p = url.searchParams;
  const amountText = p.get('am');
  const amount = amountText ? Number(amountText) : null;
  if (amountText && (!Number.isFinite(amount) || amount! < 0)) throw new Error('The amount in this UPI URI is invalid.');
  return { payee:p.get('pa')||'Not provided', name:p.get('pn')||'Not provided', amount, note:p.get('tn')||p.get('tr')||'Not provided', currency:p.get('cu')||'INR', mode:p.get('mode')||'Not specified', raw };
}

const API_BASE = (import.meta.env.VITE_API_BASE_URL || '').replace(/\/$/, '');

export type RiskLevel = 'danger' | 'caution' | 'clear';

export class CheckFirstApiError extends Error {
  status: number;
  code?: string;
  requestId?: string;
  constructor(message: string, status: number, code?: string, requestId?: string) {
    super(message); this.name = 'CheckFirstApiError'; this.status = status; this.code = code; this.requestId = requestId;
  }
}

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  if (!API_BASE) throw new CheckFirstApiError('Backend is not configured. Running in local demo mode.', 0, 'BACKEND_NOT_CONFIGURED');
  const controller = new AbortController();
  const timeout = window.setTimeout(() => controller.abort(), 20000);
  try {
    const response = await fetch(`${API_BASE}${path}`, {
      ...init,
      signal: controller.signal,
      headers: { 'Content-Type': 'application/json', ...(init?.headers || {}) },
      credentials: 'include',
    });
    const body = await response.json().catch(() => ({}));
    if (!response.ok) {
      const apiError = body as ApiError;
      throw new CheckFirstApiError(apiError.error || 'The request could not be completed.', response.status, apiError.code, apiError.requestId);
    }
    return body as T;
  } catch (error) {
    if (error instanceof CheckFirstApiError) throw error;
    if (error instanceof DOMException && error.name === 'AbortError') throw new CheckFirstApiError('The request timed out. Please try again.', 408, 'TIMEOUT');
    throw new CheckFirstApiError('Cannot reach the CheckFirst service. Check your connection and try again.', 0, 'NETWORK_ERROR');
  } finally { window.clearTimeout(timeout); }
}

=== FILE: checkfirst-frontend/README.md ===
# CheckFirst frontend starter

A responsive React + TypeScript frontend inspired by the CheckFirst Figma design. It includes eight separate pages and self-contained demo interactions.

## Pages
- Dashboard with illustrative analytics
- Scam Message Scanner (local rule-based demo)
- QR & Payment Link Analyzer (local UPI URI parsing and QR decoding)
- Pause & Verify Guide
- Emergency Help and browser-generated incident summary
- Loan Cost Calculator
- Loan Risk Checker
- Settings (language toggle, Simple Mode, larger text, opt-in local check metadata)

## Run locally
Requires Node.js 18+.

```bash
npm install
npm run dev
```

Open the URL printed by Vite (usually http://localhost:5173).

To verify production compilation:

```bash
npm run build
```

## Google AI Studio handoff
Upload this folder/project to Google AI Studio (Build mode / code workspace) and paste the supplied backend prompt. The frontend intentionally works in a local demo mode before an API is configured.

Set `VITE_API_BASE_URL` in `.env` only after your backend is deployed. Do not put Gemini API keys, Firebase Admin credentials, or other secrets in this frontend environment. All secrets belong on the server.

## Safety and demo limitations
- Message scanning is a small deterministic keyword/rule demo, not a trained scam detector.
- QR parsing reads UPI URI fields. It does not verify the payee's identity, account ownership, or trustworthiness.
- Loan results are estimates, not regulatory APR disclosures or financial advice.
- Dashboard analytics are fictional demonstration values until connected to real, consented app events.
- Incident summaries are generated locally. No data is sent to CheckFirst in this frontend-only starter.
- Optional history is off by default and stores only metadata locally, not raw message text, QR contents, PINs, or OTPs.

## Backend contract expected by the UI integration prompt
Suggested API endpoints:
- `POST /api/v1/scan/message`
- `POST /api/v1/scan/upi` (for decoded UPI URI fields; QR decoding itself can stay client-side)
- `POST /api/v1/loan/calculate`
- `POST /api/v1/loan/assess`
- `GET /api/v1/dashboard/summary`
- `GET /api/v1/history` and `DELETE /api/v1/history` (opt-in metadata only)

Do not accept or persist payment credentials, UPI PINs, OTPs, passwords, full card data, or raw QR/message content by default.
